\newcommand{\mytriangleone}[1][1]{
\begin{tikzpicture}[scale=#1]
\fill[yellow] (0,0) -- (0.5,1.4) node[above]{\textcolor{black}{A}} -- (1,0);
\fill[yellow] (0.5,-0.1) parabola (0,0) node[left]{\textcolor{black}{C}};
\fill[yellow] (0.5,-0.1) parabola (1,0) node[right]{\textcolor{black}{B}} -- (0,0) ;
\end{tikzpicture}
}

\newcommand{\mytriangletwo}[1][1]{
\begin{tikzpicture}[scale=#1]
\fill[black] (0,0) node[left]{F} -- (0.5, 1.4) node[above]{D} -- (0.8,-0.1) node[right]{E} parabola (0,0);
\end{tikzpicture}
}

\newcommand{\myline}[5]{
\begin{tikzpicture}[scale=#1]
\draw[line width=#2, #3] (0,0) node[above]{\textcolor{black}{#4}} -- (1,0) node[above]{\textcolor{black}{#5}};
\end{tikzpicture}
}

\newcommand{\mygianttriangleone}[1][1]{

\draw[blue, line width=2] (2,5) -- (3,1.5);

}

\newcommand{\mygianttriangletwo}[1][1]{

\draw[blue, line width=2] (2,6) -- (2.5,2);
\draw[yellow, line width=2] (2.5,2) -- (0,0);
\end{tikzpicture}
}
